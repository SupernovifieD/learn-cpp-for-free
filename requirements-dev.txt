# Python packages for the browser tests and the C++ formatter. The book itself is
# built by mdBook, which scripts/install_tools.py downloads (see `make install`).

# Test runner for the browser (layout, accessibility, behaviour) tests.
pytest==9.2.0

# Runs test groups in parallel (make test-ui uses it) to keep the browser tests
# fast: every page is checked at nine widths in two color schemes.
pytest-xdist==3.8.0

# Headless Chromium driver. 1.64.1 is pinned (not 1.65.x) on purpose: it uses
# Chromium revision 1248, so machines that already have that browser cached do
# not need a large download. Move to a newer release together with the cache.
playwright==1.64.1

# Runs the axe-core accessibility rules inside Playwright pages.
axe-playwright-python==0.1.8

# Pinned formatter so every contributor formats C++ examples identically.
clang-format==23.1.5
